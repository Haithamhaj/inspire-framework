## Core Components

### Button

```tsx
// src/components/ui/Button.tsx
type Variant = 'primary' | 'secondary' | 'accent' | 'ghost'

export function Button({
  children, variant = 'primary', disabled, onClick, type = 'button', className = ''
}: {
  children: React.ReactNode
  variant?: Variant
  disabled?: boolean
  onClick?: () => void
  type?: 'button' | 'submit'
  className?: string
}) {
  const base = 'px-6 py-3 rounded-lg font-medium text-sm transition-all duration-150 disabled:opacity-40 disabled:cursor-not-allowed'
  const variants = {
    primary: 'bg-primary text-white hover:bg-primary-hover',
    accent: 'bg-accent text-white hover:bg-accent-hover',
    secondary: 'border border-primary text-primary hover:bg-gray-50',
    ghost: 'text-gray-600 hover:text-primary hover:bg-gray-50',
  }
  return (
    <button type={type} disabled={disabled} onClick={onClick}
      className={`${base} ${variants[variant]} ${className}`}>
      {children}
    </button>
  )
}
```

### Input

```tsx
// src/components/ui/Input.tsx
export function Input({
  label, id, type = 'text', value, onChange, placeholder, error, dir
}: {
  label: string, id: string, type?: string
  value: string, onChange: (v: string) => void
  placeholder?: string, error?: string, dir?: 'ltr' | 'rtl'
}) {
  return (
    <div className="mb-4">
      <label htmlFor={id} className="block text-sm text-gray-600 mb-1">{label}</label>
      <input
        id={id} type={type} value={value}
        onChange={e => onChange(e.target.value)}
        placeholder={placeholder}
        dir={dir ?? (type === 'email' || type === 'password' ? 'ltr' : undefined)}
        className={`w-full border rounded-lg px-4 py-3 text-sm outline-none transition-all
          ${error
            ? 'border-red-400 ring-2 ring-red-100 focus:border-red-400'
            : 'border-gray-300 focus:border-primary focus:ring-2 focus:ring-primary/10'
          }`}
      />
      {error && <p className="mt-1 text-xs text-red-500">{error}</p>}
    </div>
  )
}
```

### CopyButton

```tsx
// src/components/ui/CopyButton.tsx
'use client'
import { useState } from 'react'

export function CopyButton({ text, label = 'نسخ' }: { text: string; label?: string }) {
  const [copied, setCopied] = useState(false)
  const copy = async () => {
    await navigator.clipboard.writeText(text)
    setCopied(true)
    setTimeout(() => setCopied(false), 2500)
  }
  return (
    <button onClick={copy}
      className="px-4 py-2 text-sm border border-gray-200 rounded-lg hover:bg-gray-50 transition-colors">
      {copied ? '✓ تم النسخ' : label}
    </button>
  )
}
```

### ProgressBar

```tsx
// RTL-aware progress bar — fills from right to left in Arabic
export function ProgressBar({ value, max, label }: { value: number; max: number; label?: string }) {
  const pct = Math.round((value / max) * 100)
  return (
    <div className="mb-6">
      <div className="h-1 bg-gray-100 rounded-full overflow-hidden">
        {/* In RTL, we use margin-right trick instead of margin-left */}
        <div
          className="h-full bg-primary rounded-full transition-all duration-300"
          style={{ width: `${pct}%` }}
        />
      </div>
      {label && (
        <div className="flex justify-between mt-1 text-xs text-gray-400">
          <span>{label}</span>
          <span>{pct}%</span>
        </div>
      )}
    </div>
  )
}
```

## RTL Rules — Apply Everywhere

| Situation | Wrong | Correct |
|---|---|---|
| Email/password inputs | `dir="rtl"` | `dir="ltr"` |
| Code blocks / System Instruction | `dir="rtl"` | `dir="ltr" text-left font-mono` |
| Back/forward chevrons | Use `→` for "next" | Use `←` for "next" in RTL |
| Grid column order | left-to-right | reverses automatically in RTL |
| flex-row | left = first | right = first in RTL |

**Always set `dir="ltr"` on:**
- Email inputs
- Password inputs
- Code/monospace content blocks
- URL displays

**Always set `dir="rtl"` (inherit from root) on:**
- Everything else — the root layout handles this

## Layout Patterns

```tsx
// Two-column grid (reverses in RTL automatically)
<div className="grid grid-cols-2 gap-4">
  <div>First (appears on RIGHT in RTL)</div>
  <div>Second (appears on LEFT in RTL)</div>
</div>

// Navigation row
<div className="flex justify-between items-center">
  <button>← السابق</button>    {/* Left side = back in RTL */}
  <button>التالي ←</button>    {/* Right side = next in RTL — wait, flip it */}
</div>
// CORRECT for RTL:
<div className="flex justify-between items-center">
  <button>التالي ←</button>   {/* This appears on LEFT = "next" direction in RTL */}
  <button>→ السابق</button>   {/* This appears on RIGHT */}
</div>
```

## Rules

- Never use `text-left` as default — `text-right` is the RTL default
- Never hardcode `margin-left` for spacing — use `gap` or `space-x` (auto-flips in RTL)
- Test every page mentally: "would this make sense if the whole page mirrored?"
- The results code block (System Instruction) must ALWAYS be `dir="ltr"` — it's English content
- Use `font-arabic` (Tajawal) for UI text, `font-mono` for code output
