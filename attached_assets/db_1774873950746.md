## The 4 Tables

```
users              ← accounts (email + password)
refresh_tokens     ← JWT refresh tokens
assessments        ← all assessment sessions and results
admin_sessions     ← admin panel sessions
```

## Complete Prisma Schema

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id                  String    @id @default(uuid())
  createdAt           DateTime  @default(now()) @map("created_at")
  updatedAt           DateTime  @updatedAt @map("updated_at")
  email               String    @unique
  name                String
  passwordHash        String    @map("password_hash")
  jobTitle            String?   @map("job_title")
  emailVerified       Boolean   @default(false) @map("email_verified")
  emailVerifyToken    String?   @map("email_verify_token")
  emailVerifyExpires  DateTime? @map("email_verify_expires")
  consentGiven        Boolean   @default(false) @map("consent_given")
  consentAt           DateTime? @map("consent_at")
  isActive            Boolean   @default(true) @map("is_active")
  lastLoginAt         DateTime? @map("last_login_at")
  plan                String    @default("free")
  assessments         Assessment[]
  refreshTokens       RefreshToken[]
  @@map("users")
}

model RefreshToken {
  id        String   @id @default(uuid())
  userId    String   @map("user_id")
  token     String   @unique
  createdAt DateTime @default(now()) @map("created_at")
  expiresAt DateTime @map("expires_at")
  revoked   Boolean  @default(false)
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  @@map("refresh_tokens")
}

model Assessment {
  id                    String    @id @default(uuid())
  createdAt             DateTime  @default(now()) @map("created_at")
  updatedAt             DateTime  @updatedAt @map("updated_at")
  userId                String    @map("user_id")
  projectName           String    @map("project_name")
  projectGoal           String    @map("project_goal")
  reportLanguage        String    @default("ar") @map("report_language")
  assessmentType        String    @default("full") @map("assessment_type")

  // Answers
  behavioralAnswers     Json?     @map("behavioral_answers")
  scenarioAnswers       Json?     @map("scenario_answers")
  openAnswer            String?   @map("open_answer")

  // Results — 8 separate sections, NEVER merge them
  inspireTable          Json?     @map("inspire_table")      // Section 1
  roleAnalysis          String?   @map("role_analysis")      // Section 2
  redLines              Json?     @map("red_lines")          // Section 3
  strengths             Json?     @map("strengths")          // Section 4
  developmentAreas      Json?     @map("development_areas")  // Section 5
  recommendations       Json?     @map("recommendations")    // Section 6
  systemInstruction     String?   @map("system_instruction") // Section 7
  quickStarters         Json?     @map("quick_starters")     // Section 8

  // AI tracking
  aiProvider            String?   @map("ai_provider")        // 'openai' | 'anthropic'
  aiModel               String?   @map("ai_model")           // 'gpt-5.4' | 'claude-sonnet-4-6'

  // Status lifecycle
  // draft → processing → completed
  //                    ↘ pending_retry → completed
  //                                    ↘ failed
  status                String    @default("draft")
  retryCount            Int       @default(0) @map("retry_count")
  nextRetryAt           DateTime? @map("next_retry_at")

  // Email + PDF
  emailSent             Boolean   @default(false) @map("email_sent")
  emailSentAt           DateTime? @map("email_sent_at")
  pdfGenerated          Boolean   @default(false) @map("pdf_generated")
  pdfUrl                String?   @map("pdf_url")

  completionTimeSeconds Int?      @map("completion_time_seconds")
  previousAssessmentId  String?   @map("previous_assessment_id")

  user                  User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  previousAssessment    Assessment?  @relation("AssessmentHistory", fields: [previousAssessmentId], references: [id])
  nextAssessments       Assessment[] @relation("AssessmentHistory")

  @@map("assessments")
}

model AdminSession {
  id        String   @id @default(uuid())
  token     String   @unique
  createdAt DateTime @default(now()) @map("created_at")
  expiresAt DateTime @map("expires_at")
  @@map("admin_sessions")
}
```

## Prisma Client Singleton

```typescript
// src/lib/db.ts
import { PrismaClient } from '@prisma/client'

const globalForPrisma = globalThis as unknown as { prisma: PrismaClient }

export const prisma =
  globalForPrisma.prisma ??
  new PrismaClient({ log: ['error'] })

if (process.env.NODE_ENV !== 'production') globalForPrisma.prisma = prisma
```

## Common Queries

```typescript
// Get user by email
const user = await prisma.user.findUnique({ where: { email } })

// Create assessment (on submit)
const assessment = await prisma.assessment.create({
  data: {
    userId,
    projectName,
    projectGoal,
    reportLanguage,
    assessmentType,
    behavioralAnswers,
    scenarioAnswers,
    openAnswer,
    status: 'processing',
    completionTimeSeconds,
  }
})

// Update with results (all 8 sections at once)
await prisma.assessment.update({
  where: { id: assessmentId },
  data: {
    inspireTable,
    roleAnalysis,
    redLines,
    strengths,
    developmentAreas,
    recommendations,
    systemInstruction,
    quickStarters,
    status: 'completed',
    aiProvider: 'openai',
    aiModel: 'gpt-5.4',
  }
})

// Get user's assessment history
const assessments = await prisma.assessment.findMany({
  where: { userId, status: 'completed' },
  orderBy: { createdAt: 'desc' },
  select: {
    id: true, projectName: true, assessmentType: true,
    createdAt: true, aiProvider: true, aiModel: true,
  }
})

// Find pending retries for cron job
const pending = await prisma.assessment.findMany({
  where: {
    status: 'pending_retry',
    nextRetryAt: { lte: new Date() },
    retryCount: { lt: 10 },
  },
  include: { user: true }
})
```

## Important Rules

- Always use `onDelete: Cascade` for user-linked records
- Never store passwords in plain text — always bcrypt hash
- `status` valid values: `'draft'` | `'processing'` | `'pending_retry'` | `'completed'` | `'failed'`
- All 8 result sections must be stored separately — never concatenate them
- `retryCount` starts at 0, increments on each background attempt
