## Decision Flow (implement exactly this)

```
User submits answers
        ↓
Save to DB (status: 'processing')
        ↓
Try GPT-5.4 — up to 3 attempts (1 + 2 retries, 2s delay each)
        ↓
   Success? → Save all 8 sections → status: 'completed' → send email
        ↓ No
Is ANTHROPIC_API_KEY set?
   Yes → Try claude-sonnet-4-6 (1 attempt)
              ↓
         Success? → Save all 8 sections → status: 'completed' → send email
              ↓ No
   No / Claude also failed
        ↓
status: 'pending_retry', retryCount: 0
nextRetryAt: now + 30 seconds
        ↓
Cron job picks it up and repeats (OpenAI first, then Claude if available)
Intervals: 30s → 1m → 2m → 5m → 5m → 10m → 10m → 30m → 30m → 1h
        ↓
After 10 failed background attempts → status: 'failed' → send apology email
```

## Implementation

```typescript
// src/lib/ai-engine.ts

import OpenAI from 'openai'
import Anthropic from '@anthropic-ai/sdk'
import { prisma } from './db'
import { buildPrompt } from './prompt-builder'
import { parseFullReport } from './report-parser'
import { sendResultsEmail, sendFailureEmail } from './email'

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY })
const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY })

const RETRY_INTERVALS_MS = [
  30_000, 60_000, 120_000, 300_000, 300_000,
  600_000, 600_000, 1_800_000, 1_800_000, 3_600_000,
]

// ─── MAIN ENTRY ──────────────────────────────────────

export async function generateReport(
  assessmentId: string,
  promptData: PromptData
): Promise<void> {
  const prompt = buildPrompt(promptData)

  // 1. Try OpenAI
  const openaiResult = await tryOpenAI(prompt)
  if (openaiResult.success) {
    await finish(assessmentId, openaiResult.text!, 'openai', 'gpt-5.4')
    return
  }

  // 2. Try Claude (if key available)
  if (process.env.ANTHROPIC_API_KEY) {
    const claudeResult = await tryClaude(prompt)
    if (claudeResult.success) {
      await finish(assessmentId, claudeResult.text!, 'anthropic', 'claude-sonnet-4-6')
      return
    }
  }

  // 3. Both failed — queue for background retry
  await prisma.assessment.update({
    where: { id: assessmentId },
    data: {
      status: 'pending_retry',
      retryCount: 0,
      nextRetryAt: new Date(Date.now() + RETRY_INTERVALS_MS[0]),
    },
  })
}

// ─── CRON JOB (called every minute) ──────────────────

export async function processRetryQueue(): Promise<void> {
  const pending = await prisma.assessment.findMany({
    where: {
      status: 'pending_retry',
      nextRetryAt: { lte: new Date() },
      retryCount: { lt: 10 },
    },
    include: { user: true },
  })

  for (const assessment of pending) {
    const count = assessment.retryCount ?? 0
    const prompt = buildPrompt({
      name: assessment.user.name,
      jobTitle: assessment.user.jobTitle ?? undefined,
      projectName: assessment.projectName,
      projectGoal: assessment.projectGoal,
      reportLanguage: assessment.reportLanguage as any,
      behavioralAnswers: assessment.behavioralAnswers as any,
      scenarioAnswers: assessment.scenarioAnswers as any,
      openAnswer: assessment.openAnswer ?? '',
    })

    const openaiResult = await tryOpenAI(prompt)
    if (openaiResult.success) {
      await finish(assessment.id, openaiResult.text!, 'openai', 'gpt-5.4')
      continue
    }

    if (process.env.ANTHROPIC_API_KEY) {
      const claudeResult = await tryClaude(prompt)
      if (claudeResult.success) {
        await finish(assessment.id, claudeResult.text!, 'anthropic', 'claude-sonnet-4-6')
        continue
      }
    }

    // Still failing
    const nextCount = count + 1
    if (nextCount >= 10) {
      await prisma.assessment.update({
        where: { id: assessment.id },
        data: { status: 'failed', retryCount: nextCount, nextRetryAt: null },
      })
      await sendFailureEmail(assessment.user.email, assessment.user.name)
    } else {
      await prisma.assessment.update({
        where: { id: assessment.id },
        data: {
          retryCount: nextCount,
          nextRetryAt: new Date(Date.now() + (RETRY_INTERVALS_MS[count] ?? 3_600_000)),
        },
      })
    }
  }
}

// ─── PROVIDER CALLERS ────────────────────────────────

async function tryOpenAI(
  prompt: string
): Promise<{ success: boolean; text?: string }> {
  for (let attempt = 1; attempt <= 3; attempt++) {
    try {
      const res = await openai.chat.completions.create({
        model: process.env.OPENAI_MODEL ?? 'gpt-5.4',
        messages: [{ role: 'user', content: prompt }],
        max_tokens: 3500,
        temperature: 0.7,
      })
      const text = res.choices[0].message.content ?? ''
      if (!text) throw new Error('Empty response')
      return { success: true, text }
    } catch (err) {
      console.error(`[OpenAI] attempt ${attempt} failed:`, err)
      if (attempt < 3) await sleep(2000 * attempt)
    }
  }
  return { success: false }
}

async function tryClaude(
  prompt: string
): Promise<{ success: boolean; text?: string }> {
  try {
    const res = await anthropic.messages.create({
      model: process.env.ANTHROPIC_MODEL ?? 'claude-sonnet-4-6',
      max_tokens: 3500,
      messages: [{ role: 'user', content: prompt }],
    })
    const text = res.content.filter(b => b.type === 'text').map(b => b.text).join('')
    if (!text) throw new Error('Empty response')
    return { success: true, text }
  } catch (err) {
    console.error('[Claude] failed:', err)
    return { success: false }
  }
}

// ─── SAVE + EMAIL ─────────────────────────────────────

async function finish(
  assessmentId: string,
  rawText: string,
  provider: string,
  model: string
): Promise<void> {
  const parsed = parseFullReport(rawText)
  await prisma.assessment.update({
    where: { id: assessmentId },
    data: {
      ...parsed,
      status: 'completed',
      aiProvider: provider,
      aiModel: model,
      retryCount: 0,
      nextRetryAt: null,
    },
  })
  sendResultsEmail(assessmentId).catch(console.error)
}

const sleep = (ms: number) => new Promise(r => setTimeout(r, ms))
```

```typescript
import { processRetryQueue } from '@/lib/ai-engine'
import { NextRequest } from 'next/server'

export async function GET(req: NextRequest) {
  // Protect the endpoint
  const secret = req.headers.get('x-cron-secret')
  if (secret !== process.env.CRON_SECRET) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 })
  }
  await processRetryQueue()
  return Response.json({ ok: true })
}
```

## Output Parser

```typescript
// src/lib/report-parser.ts

export function parseFullReport(text: string) {
  const extract = (start: string, end: string) => {
    const m = text.match(new RegExp(`${start}([\\s\\S]*?)${end}`))
    return m ? m[1].trim() : ''
  }

  const parseJsonSection = (raw: string): any => {
    // Extract bullet lines as array
    return raw.split('\n')
      .filter(l => l.trim().startsWith('•'))
      .map(l => l.replace(/^•\s*/, '').trim())
      .filter(Boolean)
  }

  const parseTable = (raw: string) => {
    return raw.split('\n')
      .filter(l => l.includes('|') && !l.startsWith('Axis'))
      .map(l => {
        const parts = l.split('|').map(p => p.trim()).filter(Boolean)
        if (parts.length < 6) return null
        return {
          axis: parts[0],
          score: parseInt(parts[1]) || 0,
          max: parseInt(parts[2]) || 6,
          percentage: parseFloat(parts[3]) || 0,
          confidence: parseInt(parts[4]) || 0,
          note: parts[5] || '',
        }
      })
      .filter(Boolean)
  }

  const rawQS = extract('===QS_START===', '===QS_END===')
  const quickStarters = rawQS
    .split('\n')
    .filter(l => /^\d\./.test(l.trim()))
    .map(l => l.replace(/^\d\.\s*[""]?/, '').replace(/[""]?\s*$/, '').trim())
    .filter(Boolean)

  return {
    inspireTable:      parseTable(extract('===TABLE_START===', '===TABLE_END===')),
    roleAnalysis:      extract('===ROLE_START===', '===ROLE_END==='),
    redLines:          parseJsonSection(extract('===REDLINES_START===', '===REDLINES_END===')),
    strengths:         parseJsonSection(extract('===STRENGTHS_START===', '===STRENGTHS_END===')),
    developmentAreas:  parseJsonSection(extract('===DEVELOPMENT_START===', '===DEVELOPMENT_END===')),
    recommendations:   extract('===RECOMMENDATIONS_START===', '===RECOMMENDATIONS_END===')
                         .split('\n').filter(l => /^\d\./.test(l.trim()))
                         .map(l => l.replace(/^\d\.\s*/, '').trim()).filter(Boolean),
    systemInstruction: extract('===SYS_START===', '===SYS_END==='),
    quickStarters:     quickStarters.length > 0 ? quickStarters : [rawQS],
  }
}
```

## Rules

- `max_tokens` must be **3500** minimum — less will truncate the 8-section output
- Never call AI from a component or client-side code
- Always log provider + model in DB for admin visibility
- If parsed sections are empty strings, still save them — don't throw errors
